\chapter{Kết luận và Hướng phát triển}
\label{chap:conclusion}

\section{Kết luận}
\label{sec:conclusion}

Khóa luận đã nghiên cứu và xây dựng thành công \textbf{ViNeoBERT} -- một mô hình
Encoder hiện đại cho tiếng Việt -- bằng cách thích nghi NeoBERT thông qua kỹ thuật
SALT được cải tiến, thay vì huấn luyện lại từ đầu. Đối chiếu với các mục tiêu đặt ra
ở Chương~\ref{chap:intro}, đề tài đã đạt được các kết quả sau:

\begin{itemize}
  \item \textbf{Về nghiên cứu kiến trúc}: Chúng tôi đã phân tích các cải tiến kiến
        trúc của NeoBERT (RoPE, SwiGLU, Pre-RMSNorm, tỷ lệ sâu--hẹp tối ưu) và xác
        định đúng hai điểm mấu chốt ảnh hưởng tới việc chuyển giao: chuẩn hóa
        Pre-RMSNorm nhạy với độ lớn của nhúng -- dẫn tới bước hiệu chỉnh chuẩn -- và
        đầu giải mã tách rời đòi hỏi khởi tạo riêng cho lớp đầu ra
        (Chương~\ref{chap:related}, \ref{chap:method}).

  \item \textbf{Về phương pháp chuyển giao}: Chúng tôi đã triển khai thuật toán SALT
        và cải tiến nó cho cặp ngôn ngữ Anh--Việt với ba đóng góp: (1)~mở rộng tập neo
        bằng cặp dịch khai thác tự động, kèm cơ chế lọc từ đồng tự khác nghĩa đặc thù
        cho ngôn ngữ hệ chữ Latin; (2)~áp dụng phép chiếu theo từng token cho cả lớp
        đầu ra tách rời, kết hợp khởi tạo độ chệch theo tần suất từ đơn và hệ số co
        trọng số $\gamma = 0{,}1$; (3)~bổ sung giai đoạn căn chỉnh đóng băng trước khi
        tiếp tục tiền huấn luyện (Chương~\ref{chap:method}).

  \item \textbf{Về kiểm chứng thực nghiệm}: Thí nghiệm so sánh có kiểm soát các phương
        án khởi tạo ở cùng ngân sách (Mục~\ref{sec:exp-bakeoff}) xác nhận từng lựa chọn
        thiết kế: hệ số co $\gamma=0{,}1$ là cần thiết -- không co hoặc co không đủ
        khiến huấn luyện mắc kẹt ở mức mất mát cao; khởi tạo SALT riêng cho hai ma trận
        vượt xa cách buộc trọng số mặc định (67,52 so với 54,09 điểm F1 trên UIT-ViQuAD
        sau 100 triệu token); và căn chỉnh đóng băng giảm khoảng 26\% perplexity, nâng
        F1 lên 72,15 với chi phí phụ trợ không đáng kể.

  \item \textbf{Về hiệu quả của mô hình}: Sau khoảng 5 tỷ token CulturaX
        ($\sim$20GB văn bản), ViNeoBERT
        đạt 76,62 điểm F1 trên đọc hiểu UIT-ViQuAD (hơn XLM-R-base 3,3 điểm), cao hơn
        XLM-R-base ở 11/12 tác vụ của bảng đánh giá đa nhiệm, dẫn đầu cả năm mô hình
        so sánh ở gán nhãn từ loại và cạnh tranh sòng phẳng với các bản \texttt{large}
        có gấp 1,5--2,3 lần tham số (Mục~\ref{sec:exp-mrc}). Pha mở rộng ngữ cảnh
        $\sim$200 triệu token ở độ dài 4.096 giữ pseudo-perplexity ổn định trên toàn
        dải ngữ cảnh, biến trần 4.096 token kế thừa từ kiến trúc thành năng lực sử
        dụng được. Đáng nói nhất, lượng tiếng Việt mô hình tiếp xúc chỉ
        khoảng 5 tỷ token -- ít hơn một đến hai bậc so với PhoBERT (40 epoch trên 20GB,
        bản v2 thêm 120GB OSCAR) hay XLM-R ($\sim$137GB tiếng Việt trong CC-100). Kết
        quả này xác nhận giả thuyết trung tâm của đề tài: \textit{tái sử dụng không
        gian biểu diễn sẵn có là con đường tiết kiệm để đưa các kiến trúc Encoder thế
        hệ mới đến với tiếng Việt}.
\end{itemize}

Bên cạnh mô hình, đề tài đóng góp một quy trình ba giai đoạn có thể tái lập -- khởi
tạo SALT cải tiến, căn chỉnh đóng băng, tiếp tục tiền huấn luyện theo lịch WSD -- cùng
bộ mã nguồn công khai đi kèm.

\section{Hạn chế của đề tài}
\label{sec:limitations}

Trong khuôn khổ thời gian và tài nguyên của một khóa luận, đề tài còn các hạn chế
sau:

\begin{itemize}
  \item \textbf{Ngân sách huấn luyện}: khoảng 5 tỷ token ($\sim$20GB văn bản) vẫn
        nhỏ so với dữ liệu tiền
        huấn luyện của các mô hình đối sánh; ViNeoBERT chưa vượt PhoBERT-base-v2 ở
        nhóm tác vụ phân loại câu ngắn, dù xu hướng đường cong học cho thấy khoảng
        cách có thể tiếp tục thu hẹp khi tăng ngân sách.
  \item \textbf{Phạm vi đánh giá}: bộ đánh giá tập trung vào các tác vụ hiểu và biểu
        diễn câu; một số năng lực như nhận dạng thực thể có tên hay ứng dụng ngữ cảnh
        dài chưa được khảo sát, và hai tác vụ truy hồi--xếp hạng lại mới chạy một lần
        nên chưa ước lượng được phương sai.
  \item \textbf{Ngữ cảnh dài}: pha mở rộng ngữ cảnh lên 4.096 token mới được kiểm
        chứng bằng độ đo nội tại (pseudo-perplexity theo độ dài chuỗi,
        Mục~\ref{subsec:exp-ctx4096}); chưa có đánh giá hạ nguồn trên các tác vụ
        đòi hỏi ngữ cảnh dài như hỏi đáp đa đoạn hay truy hồi tài liệu.
  \item \textbf{Bộ tách từ}: từ vựng kế thừa từ ViDeBERTa hoạt động ở mức âm tiết;
        ảnh hưởng của việc không tách từ (so với PhoBERT dùng dữ liệu đã tách từ)
        chưa được phân tích riêng.
\end{itemize}

\section{Hướng phát triển}
\label{sec:future-work}

Từ các kết quả và hạn chế trên, chúng tôi đề xuất năm hướng phát triển:

\begin{enumerate}
  \item \textbf{Mở rộng ngân sách tiếp tục tiền huấn luyện}: tăng lượng token CulturaX
        vượt mốc 5 tỷ và khai thác thêm các pha làm nguội trên dữ liệu mới -- vốn
        đã cho thấy hiệu quả bật điểm ở mốc cuối -- nhằm vượt các mô hình họ PhoBERT
        trên các tác vụ hạ nguồn.
  \item \textbf{Trộn lại một phần dữ liệu tiếng Anh khi tiếp tục tiền huấn luyện}: giữ
        lại một tỷ lệ nhỏ (khoảng $5$--$10\%$) dữ liệu tiếng Anh -- ngôn ngữ nguồn của
        NeoBERT -- trong hỗn hợp huấn luyện ở Giai đoạn 3 như một cơ chế \textit{phát
        lại} (replay), nhằm bảo toàn biểu diễn ở các tầng trong và giảm quên lãng thảm
        họa. Ibrahim và cộng sự~\cite{ibrahim2024continual} cho thấy chỉ cần phát lại
        một phần nhỏ dữ liệu cũ là đủ để chống suy sụp khi phân phối dữ liệu dịch chuyển;
        đáng chú ý hơn, Elhady và cộng sự~\cite{elhady2025emergent} chỉ ra rằng trong
        thích nghi ngôn ngữ, việc giữ lại dữ liệu tiếng Anh tuy \textit{không} cải thiện
        perplexity nhưng lại quyết định sự xuất hiện của năng lực hạ nguồn -- một tín hiệu
        mà mất mát MLM đơn thuần không phản ánh. Đề tài dùng khởi tạo SALT và giai đoạn
        căn chỉnh đóng băng làm lớp bảo vệ chống quên lãng, còn phát lại tiếng Anh là một
        cơ chế bổ trợ tự nhiên đáng để khảo sát tiếp.
  \item \textbf{Khai thác ngữ cảnh dài ở hạ nguồn}: xây dựng và đánh giá trên các tác
        vụ đòi hỏi ngữ cảnh dài (hỏi đáp đa đoạn, truy hồi tài liệu) để bổ sung cho
        kiểm chứng nội tại bằng pseudo-perplexity ở Mục~\ref{subsec:exp-ctx4096}; khi
        cần vượt trần 4.096 token, các kỹ thuật nội suy RoPE như
        YaRN~\cite{peng2023yarn} là ứng viên tự nhiên.
  \item \textbf{Mở rộng bộ đánh giá}: bổ sung nhận dạng thực thể có tên, truy hồi
        trên kho ngữ liệu quy mô lớn hơn và lặp lại nhóm tác vụ biểu diễn câu qua
        nhiều lần chạy, nhằm đánh giá toàn diện và tin cậy hơn chất lượng biểu diễn
        của mô hình ở vai trò một Encoder đa dụng.
  \item \textbf{Tổng quát hóa quy trình}: áp dụng quy trình ba giai đoạn của đề tài
        cho các ngôn ngữ ít tài nguyên khác hoặc các kiến trúc nền khác, qua đó kiểm
        chứng tính tổng quát của các cải tiến (tập neo theo cặp dịch, khởi tạo lớp
        đầu ra tách rời, căn chỉnh đóng băng) ngoài phạm vi cặp Anh--Việt và NeoBERT.
\end{enumerate}
