\chapter{Thực nghiệm và Đánh giá}
\label{chap:experiment}

\section{Môi trường và dữ liệu thực nghiệm}
\begin{itemize}
    \item Cấu hình phần cứng và phần mềm.
    \item Bộ dữ liệu tiếng Việt sử dụng để tinh chỉnh và đánh giá (Ví dụ: dữ liệu phân loại, hỏi đáp, STS).
\end{itemize}

\section{Thiết lập các tham số huấn luyện (Hyperparameters)}
\begin{itemize}
    \item Batch size, learning rate, số epoch, tối ưu hóa (Optimizer).
\end{itemize}

\section{Kịch bản đánh giá}
\begin{itemize}
    \item Các Baseline Models (ViDeBERTa gốc, phobert, v.v.).
    \item Các tiêu chí đánh giá (MTEB benchmark cho tiếng Việt, Độ chính xác, F1-score, Tốc độ hội tụ).
\end{itemize}

\section{Kết quả và Thảo luận}
\begin{itemize}
    \item Hiệu năng trên các tác vụ hiểu ngôn ngữ (NLU).
    \item Khả năng xử lý ngữ cảnh dài (Long-context capability).
    \item So sánh thời gian và chi phí tính toán (Giảm thiểu do tái sử dụng trọng số).
    \item Phân tích lỗi (Error Analysis).
\end{itemize}
